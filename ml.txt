#bisection


function [x] = bisection(a,b,m)

f = @(x) x^3-x-1;

format long

c = f(a);
d = f(b);

if c*d > 0
    error('Function has same sign at both ends')
end

disp(['x       y'])

for i = 1:m

    x = (a+b)/2;
    y = f(x);

    disp([x y])

    if y == 0
        break
    end

    if c*y < 0
        b = x;
    else
        a = x;
    end

end

end




#regulafalsi


function [x] = regulafalsi(a,b,m)

f = @(x) x^3-x-1;

format long

c = f(a);
d = f(b);

if c*d > 0
    error('Function has same sign at both ends')
end

disp(['i       x       y'])

for i = 1:m

    x = (a*f(b)-b*f(a))/(f(b)-f(a));

    y = f(x);

    disp([i x y])

    if y == 0
        break
    end

    if c*y < 0
        b = x;
    else
        a = x;
    end

end

end


#newtonrampson


function [x] = newtonraphson(x,m)

f = @(x) x^3-x-1;
df = @(x) 3*x^2-1;

format long

disp(['x       y'])

for i = 1:m

    y = f(x);

    disp([x y])

    if abs(y) < 1e-12
        break
    end

    x = x-y/df(x);

end

end

4.trapezodial


function [IT] = trapezoidal_rule(a,b,n)

f = @(x) 1/(1+x^2);

h = (b-a)/n;

s = 0;

for i = 1:n-1

    s = s+f(a+i*h);

end

IT = h*(f(a)+f(b)+2*s)/2;

end




5. simpson 1/3 rule



function [IT] = simpson13(a,b,n)

f = @(x) 1/(1+x^2);

if mod(n,2) ~= 0
    error('n must be even')
end

h = (b-a)/n;

s = 0;
t = 0;

for i = 1:2:n-1

    s = s+f(a+i*h);

end

for i = 2:2:n-2

    t = t+f(a+i*h);

end

IT = h*(f(a)+f(b)+4*s+2*t)/3;

end



6.simpson 3/8 rule


function [IT] = simpson38(a,b,n)

f = @(x) 1/(1+x^2);

if mod(n,3) ~= 0
    error('n must be a multiple of 3')
end

h = (b-a)/n;

s = 0;
t = 0;

for i = 1:n-1

    if mod(i,3) ~= 0
        s = s+f(a+i*h);
    end

end

for i = 3:3:n-3

    t = t+f(a+i*h);

end

IT = (3*h/8)*(f(a)+f(b)+3*s+2*t);

end



7. newton forward


function [p] = newton_forward(x,y,a)

m = length(x);

diff = zeros(m,m);

diff(:,1) = y(:);

for j = 2:m

    for i = 1:m-j+1

        diff(i,j) = diff(i+1,j-1)-diff(i,j-1);

    end

end

s = (a-x(1))/(x(2)-x(1));

t = s;

p = diff(1,1);

for j = 2:m

    t = t*(s-j+2)/(j-1);

    p = p+t*diff(1,j);

end

disp(diff)

end



8. Newton backward



function [p] = newton_backward(x,y,a)

m = length(x);

diff = zeros(m,m);

diff(:,1) = y(:);

for j = 2:m

    for i = j:m

        diff(i,j) = diff(i,j-1)-diff(i-1,j-1);

    end

end

s = (a-x(m))/(x(2)-x(1));

t = s;

p = diff(m,1);

for j = 2:m

    t = t*(s+j-2)/(j-1);

    p = p+t*diff(m,j);

end

disp(diff)

end



9. lagrange





function [l] = lagrange(x,y,a)

m = length(x);

for i = 1:m

    w(i) = 1;

    for j = 1:m

        if i ~= j

            w(i) = w(i)*(a-x(j))/(x(i)-x(j));

        end

    end

end

b = 0;

for i = 1:m

    b = b+w(i)*y(i);

end

l = b;

end


10. divided interpolation 




function [p] = divided(x,y,a)

m = length(x);

d = zeros(m,m);

for i = 1:m

    d(i,1) = y(i);

end

for j = 2:m

    for i = 1:m-j+1

        d(i,j) = (d(i+1,j-1)-d(i,j-1))/(x(i+j-1)-x(i));

    end

end

b = d(1,1);

t = 1;

for j = 2:m

    t = t*(a-x(j-1));

    b = b+t*d(1,j);

end

p = b;

disp(d)

end



11. taylor series



function [x,y] = taylor_series(x0,y0,a,h)

m = (a-x0)/h;

disp(['x       y'])

for i = 1:m

    y1 = x0^2+y0^2;

    y2 = 2*x0+2*y0*y1;

    y3 = 2+2*(y0*y2+y1^2);

    y4 = 2*(y0*y3+3*y1*y2);

    y = y0+h*y1+(h^2/2)*y2+(h^3/6)*y3+(h^4/24)*y4;

    x = x0+h;

    fprintf('%0.1f    %0.5f\n',x,y)

    x0 = x;
    y0 = y;

end

end

function [ys,yi,ym] = euler_method(x0,y0,m,h)

ys = y0;
yi = y0;
ym = y0;

f = @(x,y) (2-y^2)/(5*x);

disp('x     simple    improved    modified')

for i = 1:m

    ys = ys+h*f(x0,ys);

    yi = yi+(h/2)*(f(x0,yi)+f(x0+h,yi+h*f(x0,yi)));

    ym = ym+h*f(x0+h/2,ym+(h/2)*f(x0,ym));

    x = x0+h;

    fprintf('%0.1f   %0.10f   %0.10f   %0.10f\n',x,ys,yi,ym)

    x0 = x;

end

end



13. ranga kutta



function [x,y] = runge_kutta(x0,y0,h,m)

f = @(x,y) (y^2-x)/y;

disp(['x       y'])

for i = 1:m

    k1 = h*f(x0,y0);

    k2 = h*f(x0+h/2,y0+k1/2);

    k3 = h*f(x0+h/2,y0+k2/2);

    k4 = h*f(x0+h,y0+k3);

    k = (k1+2*k2+2*k3+k4)/6;

    x = x0+h;

    y = y0+k;

    disp([x y])

    x0 = x;
    y0 = y;

end

end



14. gauss elimination



function [x] = gauss(a,b)

[m,n] = size(a);

if m ~= n
    error('A is not a square matrix')
end

for k = 1:n-1

    for i = k+1:n

        a(i,:) = a(i,:)-(a(i,k)/a(k,k))*a(k,:);

        b(i) = b(i)-(a(i,k)/a(k,k))*b(k);

    end

end

x(n) = b(n)/a(n,n);

for k = n-1:-1:1

    s = 0;

    for j = k+1:n

        s = s+a(k,j)*x(j);

    end

    x(k) = (b(k)-s)/a(k,k);

end

x = x(:);

end



15. gauss seidal


function [x] = gauss_seidel(a,b,l)

m = size(a,1);

for i = 1:m

    s = 0;

    for j = 1:m

        if i ~= j

            s = s+abs(a(i,j));

        end

    end

    if abs(a(i,i)) <= s
        error('Solution may not converge')
    end

end

for i = 1:m

    x(i) = 0;

end

for k = 1:l

    old = x;

    for i = 1:m

        c = b(i);

        for j = 1:m

            if i ~= j

                c = c-a(i,j)*x(j);

            end

        end

        x(i) = c/a(i,i);

    end

    if max(abs(x-old)) < 1e-10
        break
    end

end

x = x(:);

end

function [x] = jacobi(a,b,l)

m = size(a,1);

for i = 1:m

    s = 0;

    for j = 1:m

        if i ~= j

            s = s+abs(a(i,j));

        end

    end

    if abs(a(i,i)) <= s
        error('Solution may not converge')
    end

end

for i = 1:m

    x(i) = 0;

end

for k = 1:l

    for i = 1:m

        c(i) = b(i);

        for j = 1:m

            if i ~= j

                c(i) = c(i)-a(i,j)*x(j);

            end

        end

    end

    for i = 1:m

        xnew(i) = c(i)/a(i,i);

    end

    if max(abs(xnew-x)) < 1e-10
        x = xnew;
        break
    end

    x = xnew;

end

x = x(:);

end
